cmake_minimum_required(VERSION 3.21)
project(KisakDamageDispatchRegression LANGUAGES CXX)
get_filename_component(REPO "${CMAKE_CURRENT_LIST_DIR}/../../.." ABSOLUTE)
if(NOT DEFINED GENERATED_SOURCE)
  message(FATAL_ERROR "Run tools/re/test_damage_dispatch.py")
endif()
add_executable(damage_dispatch main.cpp "${GENERATED_SOURCE}")
add_library(damage_compile_options INTERFACE)
add_library(damage_original_layout OBJECT EXCLUDE_FROM_ALL original_layout.cpp)
add_library(cgame_original_layout OBJECT EXCLUDE_FROM_ALL original_cgame_layout.cpp)
target_link_libraries(cgame_original_layout PRIVATE damage_compile_options)
target_link_libraries(damage_original_layout PRIVATE damage_compile_options)
target_include_directories(damage_original_layout PRIVATE "${CMAKE_CURRENT_BINARY_DIR}")
target_link_libraries(damage_dispatch PRIVATE damage_compile_options)
target_include_directories(damage_compile_options INTERFACE "${REPO}" "${REPO}/src" "${REPO}/src/libs" "${REPO}/src/universal")
target_compile_features(damage_compile_options INTERFACE cxx_std_20)
target_compile_definitions(damage_compile_options INTERFACE NDEBUG KISAK_MP)
if(EMSCRIPTEN)
  target_compile_definitions(damage_compile_options INTERFACE KISAK_WEB)
  target_include_directories(damage_compile_options BEFORE INTERFACE "${REPO}/src/web/compat")
  target_compile_options(damage_compile_options INTERFACE -msimd128 -msse2 -fms-extensions -fms-compatibility -fms-compatibility-version=19.40 -fdeclspec -Wno-everything "SHELL:-include ${REPO}/src/web/compat/kisak_web_compat.h")
  target_compile_options(damage_dispatch PRIVATE -O2 -fwrapv -fno-strict-aliasing -ffp-contract=off)
  target_link_options(damage_dispatch PRIVATE -O2 -sENVIRONMENT=node -sEXIT_RUNTIME=1)
else()
  if(NOT MSVC OR NOT CMAKE_SIZEOF_VOID_P EQUAL 4)
    message(FATAL_ERROR "Use MSVC x86")
  endif()
  target_compile_definitions(damage_compile_options INTERFACE WIN32 _CRT_SECURE_NO_WARNINGS)
  target_compile_options(damage_dispatch PRIVATE /O2 /fp:precise /permissive)
endif()
